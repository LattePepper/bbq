        <h2>📋 採購總數清單</h2>

<script>

    // 你提供嘅 BBQ 食物清單

    const foods = [

        "雞翼", "豬扒", "雞扒", "牛仔骨", "牛扒", 

        "豬頸肉", "金沙骨", "魷魚", "羊架", 

        "廚師腸", "芝士腸", "麵包", "多春魚"

    ];


    const foodGrid = document.getElementById('foodGrid');

    const summaryList = document.getElementById('summary-list');


    // 初始化介面

    foods.forEach((food, index) => {

        const itemDiv = document.createElement('div');

        itemDiv.className = 'food-item';

        itemDiv.innerHTML = `

            <span class="food-name">${food}</span>

            <div class="controls">

                <button class="btn-ctrl" onclick="updateQty(${index}, -1)">-</button>

                <input type="number" id="qty-${index}" class="qty-input" value="0" min="0" oninput="updateSummary()">

                <button class="btn-ctrl" onclick="updateQty(${index}, 1)">+</button>

            </div>

        `;

        foodGrid.appendChild(itemDiv);

    });


    // 改變數量函數

    function updateQty(index, change) {

        const input = document.getElementById(`qty-${index}`);

        let currentVal = parseInt(input.value) || 0;

        let newVal = currentVal + change;

        if (newVal < 0) newVal = 0;

        input.value = newVal;

        updateSummary();

    }


    // 更新總數顯示

    function updateSummary() {

        summaryList.innerHTML = '';

        let totalCount = 0;

        let hasItem = false;


        foods.forEach((food, index) => {

            const qty = parseInt(document.getElementById(`qty-${index}`).value) || 0;

            if (qty > 0) {

                hasItem = true;

                totalCount += qty;

                const li = document.createElement('li');

                li.innerHTML = `<b>${food}</b>：${qty} 件 / 份`;

                summaryList.appendChild(li);

            }

        });


        if (!hasItem) {

            summaryList.innerHTML = '<li>暫時未選擇任何食物</li>';

        } else {

            const totalLi = document.createElement('li');

            totalLi.style.marginTop = '10px';

            totalLi.style.color = '#d9534f';

            totalLi.innerHTML = `<b>總合件數/份數：${totalCount}</b>`;

            summaryList.appendChild(totalLi);

        }

    }


    // 複製清單到剪貼簿

    function copySummary() {

        let text = "🔥 BBQ 採購清單 🔥\n";

        let hasItem = false;

        let totalCount = 0;


        foods.forEach((food, index) => {

            const qty = parseInt(document.getElementById(`qty-${index}`).value) || 0;

            if (qty > 0) {

                hasItem = true;

                totalCount += qty;

                text += `- ${food}: ${qty}\n`;

            }

        });


        if (!hasItem) {

            alert("目前未有選擇任何食物哦！");

            return;

        }


        text += `\n總合件數/份數：${totalCount}`;


        navigator.clipboard.writeText(text).then(() => {

            alert("✅ 已經成功複製清單！可以直接去 WhatsApp 貼上！");

        }).catch(err => {

            alert("❌ 複製失敗，請手動複製。");

        });

    }

</script>
